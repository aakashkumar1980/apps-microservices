/** ═══════════════════════════════════════════════════════════

 *  - Only fields that client can send

public record CampaignRequest(
    String name,
    String status,
    LocalDate startDate,
    LocalDate endDate
) {}

/** Usage:
 *  CampaignRequest req = new CampaignRequest("Sale", "ACTIVE", ...);
 *  String name = req.name();  // auto-generated getter
 */


/** ═══════════════════════════════════════════════════════════
 *  Response DTO - Used for API response
 *  - Only fields that client needs to see
 *  - May hide sensitive/internal fields

public record CampaignResponse(
    String id,
    String name,
    String status,
    LocalDate startDate,
    LocalDate endDate
) {

            entity.getStatus(),
            entity.getStartDate(),
            entity.getEndDate()
        );
    }
}

🎯 REST Controller
/** ═══════════════════════════════════════════════════════════
 *  REST Controller Layer
 *  - Handles HTTP requests
 *  - Returns ResponseEntity for HTTP status control
 *  ═══════════════════════════════════════════════════════════ */

@RestController
@RequestMapping("/api/v1/campaigns")
@RequiredArgsConstructor
public class CampaignController {

  private final CampaignService service;
  
  /** ═══════════════════════════════════════════════════════════
   *  GET All Campaigns
   *  - Returns: 200 OK with list
   *  ═══════════════════════════════════════════════════════════ */
  @GetMapping
  public ResponseEntity<List<Campaign>> getAllCampaigns() {
    List<Campaign> campaigns = service.findAll();
    return ResponseEntity.ok(campaigns);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  GET Campaign by ID (PathVariable)
   *  - @PathVariable: extracts {id} from URL path
   *  - Returns: 200 OK or 404 NOT_FOUND
   *  ═══════════════════════════════════════════════════════════ */
  @GetMapping("/{id}")
  public ResponseEntity<Campaign> getCampaignById(
      @PathVariable String id) {
    Campaign campaign = service.findById(id);
    return ResponseEntity.ok(campaign);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  GET Campaigns by Status (RequestParam)
   *  - @RequestParam: extracts ?status=ACTIVE from query string
   *  - Example: /api/v1/campaigns?status=ACTIVE
   *  ═══════════════════════════════════════════════════════════ */
  @GetMapping(params = "status")
  public ResponseEntity<List<Campaign>> getCampaignsByStatus(
      @RequestParam String status) {
    List<Campaign> campaigns = service.findByStatus(status);
    return ResponseEntity.ok(campaigns);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  POST Create Campaign
   *  - @RequestBody: parses JSON body into object
   *  - Returns: 201 CREATED with created entity
   *  ═══════════════════════════════════════════════════════════ */
  @PostMapping
  public ResponseEntity<Campaign> createCampaign(
      @RequestBody Campaign request) {
    Campaign created = service.create(request);
    return ResponseEntity
      .status(HttpStatus.CREATED)
      .body(created);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  PUT Update Campaign
   *  - @PathVariable: extracts {id} from URL
   *  - @RequestBody: parses JSON body
   *  - Returns: 200 OK with updated entity
   *  ═══════════════════════════════════════════════════════════ */
  @PutMapping("/{id}")
  public ResponseEntity<Campaign> updateCampaign(
      @PathVariable String id,
      @RequestBody Campaign request) {
    Campaign updated = service.update(id, request);
    return ResponseEntity.ok(updated);
  }
}

⚙️ Service Layer
/** ═══════════════════════════════════════════════════════════
 *  Service Layer
 *  - Business logic
 *  - Calls repository for data access
 *  ═══════════════════════════════════════════════════════════ */

@Service
@RequiredArgsConstructor
public class CampaignService {

  private final CampaignRepository repository;
  
  /** GET All - calls repository.findAll() */
  public List<Campaign> findAll() {
    return repository.findAll();
  }
  
  /** GET by ID - calls repository.findById() */
  public Campaign findById(String id) {
    return repository.findById(id)
      .orElseThrow(() -> new RuntimeException(
        "Campaign not found: " + id));
  }
  
  /** GET by Status - calls repository.findByStatus() */
  public List<Campaign> findByStatus(String status) {
    return repository.findByStatus(status);
  }
  
  /** POST Create - calls repository.save() */
  public Campaign create(Campaign request) {
    Campaign campaign = Campaign.builder()
      .id(generateId())
      .name(request.getName())
      .status(request.getStatus())
      .startDate(request.getStartDate())
      .endDate(request.getEndDate())
      .build();
      
    return repository.save(campaign);
  }
  
  /** PUT Update - calls repository.save() */
  public Campaign update(String id, Campaign request) {
    /** 1. Fetch existing or throw exception */
    Campaign existing = findById(id);
    
    /** 2. Update fields */
    existing.setName(request.getName());
    existing.setStatus(request.getStatus());
    existing.setStartDate(request.getStartDate());
    existing.setEndDate(request.getEndDate());
    
    /** 3. Save and return */
    return repository.save(existing);
  }
  
  private String generateId() {
    return "C" + System.currentTimeMillis();
  }
}

🗃️ JPA Entities (Data Models)
SQL Entities (JPA) 
/** ═══════════════════════════════════════════════════════════
 *  JPA Entity - Maps to SQL Table
 *  - @Entity marks class as JPA entity
 *  - @Table specifies table name
 *  - Getters/setters omitted for brevity
 *  ═══════════════════════════════════════════════════════════ */

@Entity
@Table(name = "campaign")
public class Campaign {

    @Id
    private String id;
    
    @Column(name = "name", nullable = false)
    private String name;
    
    @Column(name = "status")
    private String status;
    
    @Column(name = "start_date")
    private LocalDate startDate;
    
    @Column(name = "end_date")
    private LocalDate endDate;
    
    /** One Campaign has Many Offers (1:N relationship) */
    @OneToMany(mappedBy = "campaign", fetch = FetchType.LAZY)
    private List<Offer> offers;
}


/** ═══════════════════════════════════════════════════════════
 *  Offer Entity - Has ManyToOne relationship with Campaign
 *  - @ManyToOne: Many Offers belong to One Campaign
 *  - @JoinColumn: Specifies FK column name
 *  ═══════════════════════════════════════════════════════════ */

@Entity
@Table(name = "offers")
public class Offer {

    @Id
    private String id;
    
    /** ═══════════════════════════════════════════════════════════
     *  ManyToOne Relationship: Offer → Campaign
     *  - This is an OBJECT reference (not just String campaignId)
     *  - Enables: findByCampaign_Id(), findByCampaign_Status()
     *  ═══════════════════════════════════════════════════════════ */
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "campaign_id")
    private Campaign campaign;  /** Object, NOT String! */
    
    @Column(name = "merchant")
    private String merchant;
    
    @Column(name = "discount")
    private Double discount;
    
    /** One Offer has Many Redemptions */
    @OneToMany(mappedBy = "offer", fetch = FetchType.LAZY)
    private List<Redemption> redemptions;
}


/** ═══════════════════════════════════════════════════════════
 *  Redemption Entity
 *  ═══════════════════════════════════════════════════════════ */

@Entity
@Table(name = "redemption")
public class Redemption {

    @Id
    private String id;
    
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "offer_id")
    private Offer offer;  /** Object reference */
    
    @Column(name = "customer_id")
    private String customerId;
    
    @Column(name = "amount")
    private Double amount;
}

NoSQL Documents (Couchbase) 
/** ═══════════════════════════════════════════════════════════
 *  Couchbase Document - Maps to JSON Document
 *  - @Document marks class as Couchbase document
 *  - @Field for explicit field naming
 *  - Uses String FK (not object) - NoSQL is denormalized
 *  ═══════════════════════════════════════════════════════════ */

@Document
public class Campaign {

    @Id
    private String id;
    
    @Field
    private String type = "campaign";  /** Type discriminator */
    
    @Field
    private String name;
    
    @Field
    private String status;
    
    @Field
    private LocalDate startDate;
    
    @Field
    private LocalDate endDate;
    
}


@Document
public class Offer {

    @Id
    private String id;
    
    @Field
    private String type = "offer";
    
    /** ═══════════════════════════════════════════════════════════
     *  NoSQL uses String FK (NOT object reference)
     *  - Documents are denormalized
     *  - No JPA relationships in NoSQL
     *  ═══════════════════════════════════════════════════════════ */
    @Field
    private String campaignId;  /** String FK, NOT object! */
    
    @Field
    private String merchant;
    
    @Field
    private Double discount;
    
}


@Document
public class Redemption {

    @Id
    private String id;
    
    @Field
    private String type = "redemption";
    
    @Field
    private String offerId;  /** String FK */
    
    @Field
    private String customerId;
    
    @Field
    private Double amount;
    
}

/* ═══════════════════════════════════════════════
 *            REDEMPTION REPOSITORY
 * ═══════════════════════════════════════════════ */
@Repository
public interface RedemptionRepository 
    extends CouchbaseRepository<Redemption, String> {

  /** ✅ SAME AS SQL - Own field queries:
   *  - findByCustomerId(String custId)
   *  - findByAmount(Double amount)
   *  - findByAmountGreaterThan(Double min)
   */

  /** ⚠️ DIFFERENT - No underscore (String FK)
   *  Entity: private String offerId; (String)
   */
  List<Redemption> findByOfferId(String offerId);

  /** ✅ SAME - Combined: FK + Own Field
   *  - findByOfferIdAndCustomerId(offerId, custId)
   *  - countByOfferId(String offerId)
   */

  /** ❌ Cannot do deep nesting - need N1QL JOINs */
  @Query("SELECT r.* FROM redemption r 
         JOIN offers o ON r.offerId = o.id
         JOIN campaign c ON o.campaignId = c.id
         WHERE r.type='redemption' AND o.type='offer' 
         AND c.type='campaign' AND c.id = $1")
  List<Redemption> findByCampaignId(String campId);

  @Query("SELECT r.* FROM redemption r 
         JOIN offers o ON r.offerId = o.id
         JOIN campaign c ON o.campaignId = c.id
         WHERE r.type='redemption' AND o.type='offer' 
         AND c.type='campaign' AND c.status = $1")
  List<Redemption> findByCampaignStatus(String s);

  @Query("SELECT COUNT(r.id) FROM redemption r 
         JOIN offers o ON r.offerId = o.id
         JOIN campaign c ON o.campaignId = c.id
         WHERE r.type='redemption' AND o.type='offer' 
         AND c.type='campaign' AND c.id = $1")
  long countByCampaignId(String campaignId);

  /** N1QL Aggregations - Same logic as SQL JPQL */
  @Query("SELECT c.id AS campaignId, c.name AS campaignName,
           COUNT(r.id) AS totalRedemptions,
           SUM(r.amount) AS totalAmount,
           AVG(r.amount) AS averageAmount,
           MAX(r.amount) AS maxAmount
         FROM redemption r JOIN offers o ON r.offerId=o.id
         JOIN campaign c ON o.campaignId=c.id
         WHERE r.type='redemption' AND o.type='offer' 
         AND c.type='campaign' AND c.status = $1
         GROUP BY c.id, c.name HAVING COUNT(r.id) > $2
         ORDER BY SUM(r.amount) DESC")
  List<CampaignStats> getCampaignStatsByStatus(
      String status, Long minCount);
}
