🎯 REST Controller
/** ═══════════════════════════════════════════════════════════
 *  REST Controller Layer
 *  - Handles HTTP requests
 *  - Returns ResponseEntity for HTTP status control
 *  ═══════════════════════════════════════════════════════════ */

@RestController
@RequestMapping("/api/v1/campaigns")
@RequiredArgsConstructor
public class CampaignController {

  private final CampaignService service;
  
  /** ═══════════════════════════════════════════════════════════
   *  GET All Campaigns
   *  - Returns: 200 OK with list
   *  ═══════════════════════════════════════════════════════════ */
  @GetMapping
  public ResponseEntity<List<Campaign>> getAllCampaigns() {
    List<Campaign> campaigns = service.findAll();
    return ResponseEntity.ok(campaigns);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  GET Campaign by ID (PathVariable)
   *  - @PathVariable: extracts {id} from URL path
   *  - Returns: 200 OK or 404 NOT_FOUND
   *  ═══════════════════════════════════════════════════════════ */
  @GetMapping("/{id}")
  public ResponseEntity<Campaign> getCampaignById(
      @PathVariable String id) {
    Campaign campaign = service.findById(id);
    return ResponseEntity.ok(campaign);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  GET Campaigns by Status (RequestParam)
   *  - @RequestParam: extracts ?status=ACTIVE from query string
   *  - Example: /api/v1/campaigns?status=ACTIVE
   *  ═══════════════════════════════════════════════════════════ */
  @GetMapping(params = "status")
  public ResponseEntity<List<Campaign>> getCampaignsByStatus(
      @RequestParam String status) {
    List<Campaign> campaigns = service.findByStatus(status);
    return ResponseEntity.ok(campaigns);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  POST Create Campaign
   *  - @RequestBody: parses JSON body into object
   *  - Returns: 201 CREATED with created entity
   *  ═══════════════════════════════════════════════════════════ */
  @PostMapping
  public ResponseEntity<Campaign> createCampaign(
      @RequestBody Campaign request) {
    Campaign created = service.create(request);
    return ResponseEntity
      .status(HttpStatus.CREATED)
      .body(created);
  }
  
  /** ═══════════════════════════════════════════════════════════
   *  PUT Update Campaign
   *  - @PathVariable: extracts {id} from URL
   *  - @RequestBody: parses JSON body
   *  - Returns: 200 OK with updated entity
   *  ═══════════════════════════════════════════════════════════ */
  @PutMapping("/{id}")
  public ResponseEntity<Campaign> updateCampaign(
      @PathVariable String id,
      @RequestBody Campaign request) {
    Campaign updated = service.update(id, request);
    return ResponseEntity.ok(updated);
  }
}

⚙️ Service Layer
/** ═══════════════════════════════════════════════════════════
 *  Service Layer
 *  - Business logic
 *  - Calls repository for data access
 *  ═══════════════════════════════════════════════════════════ */

@Service
@RequiredArgsConstructor
public class CampaignService {

  private final CampaignRepository repository;
  
  /** GET All - calls repository.findAll() */
  public List<Campaign> findAll() {
    return repository.findAll();
  }
  
  /** GET by ID - calls repository.findById() */
  public Campaign findById(String id) {
    return repository.findById(id)
      .orElseThrow(() -> new RuntimeException(
        "Campaign not found: " + id));
  }
  
  /** GET by Status - calls repository.findByStatus() */
  public List<Campaign> findByStatus(String status) {
    return repository.findByStatus(status);
  }
  
  /** POST Create - calls repository.save() */
  public Campaign create(Campaign request) {
    Campaign campaign = Campaign.builder()
      .id(generateId())
      .name(request.getName())
      .status(request.getStatus())
      .startDate(request.getStartDate())
      .endDate(request.getEndDate())
      .build();
      
    return repository.save(campaign);
  }
  
  /** PUT Update - calls repository.save() */
  public Campaign update(String id, Campaign request) {
    /** 1. Fetch existing or throw exception */
    Campaign existing = findById(id);
    
    /** 2. Update fields */
    existing.setName(request.getName());
    existing.setStatus(request.getStatus());
    existing.setStartDate(request.getStartDate());
    existing.setEndDate(request.getEndDate());
    
    /** 3. Save and return */
    return repository.save(existing);
  }
  
  private String generateId() {
    return "C" + System.currentTimeMillis();
  }
}

📦 Repository Layer
SQL (JPA Repository) 
/** ═══════════════════════════════════════════════════════════
 *  JPA Repository for SQL Database
 *  - Extends JpaRepository for built-in CRUD operations
 *  - Derived queries from method names
 *  ═══════════════════════════════════════════════════════════ */

@Repository
public interface CampaignRepository 
    extends JpaRepository<Campaign, String> {

  /** ═══════════════════════════════════════════════════════════
   *  Built-in methods from JpaRepository:
   *  - findAll()     → SELECT * FROM campaign
   *  - findById(id)  → SELECT * FROM campaign WHERE id = ?
   *  - save(entity)  → INSERT or UPDATE
   *  - delete(id)    → DELETE FROM campaign WHERE id = ?
   *  ═══════════════════════════════════════════════════════════ */
  
  /** ═══════════════════════════════════════════════════════════
   *  Derived Query - JPA generates SQL from method name
   *  - findByStatus("ACTIVE") 
   *  - Generates: SELECT * FROM campaign WHERE status = 'ACTIVE'
   *  ═══════════════════════════════════════════════════════════ */
  List<Campaign> findByStatus(String status);
  
  /** ═══════════════════════════════════════════════════════════
   *  Custom JPQL Query (optional)
   *  - Use @Query when derived query is complex
   *  ═══════════════════════════════════════════════════════════ */
  @Query("SELECT c FROM Campaign c 
         WHERE c.status = :status 
         AND c.startDate >= :date")
  List<Campaign> findActiveAfterDate(
    @Param("status") String status,
    @Param("date") LocalDate date);
}
NoSQL (Couchbase Repository) 
/** ═══════════════════════════════════════════════════════════
 *  Couchbase Repository for NoSQL Database
 *  - Extends CouchbaseRepository for built-in CRUD
 *  - Uses N1QL queries with type filter
 *  ═══════════════════════════════════════════════════════════ */

@Repository
public interface CampaignRepository 
    extends CouchbaseRepository<Campaign, String> {

  /** ═══════════════════════════════════════════════════════════
   *  Built-in methods from CouchbaseRepository:
   *  - findAll()     → SELECT * FROM bucket WHERE type='campaign'
   *  - findById(id)  → SELECT * FROM bucket USE KEYS ['id']
   *  - save(entity)  → UPSERT INTO bucket
   *  - delete(id)    → DELETE FROM bucket USE KEYS ['id']
   *  ═══════════════════════════════════════════════════════════ */

  /** ═══════════════════════════════════════════════════════════
   *  N1QL Query with type filter
   *  - Generates: SELECT * FROM bucket 
   *               WHERE type = 'campaign' AND status = 'ACTIVE'
   *  ═══════════════════════════════════════════════════════════ */
  @Query("#{#n1ql.selectEntity} 
         WHERE type = 'campaign' 
         AND status = $1")
  List<Campaign> findByStatus(String status);
}

✅ REST API Standards Followed:
• @GetMapping - Read operations (findAll, findById, findByStatus)
• @PostMapping - Create operations (save new entity)
• @PutMapping - Update operations (update existing entity)
• ResponseEntity for HTTP status control (200 OK, 201 CREATED)
• @PathVariable - Extract ID from URL path (/{id})
• @RequestParam - Extract query parameters (?status=ACTIVE)
• @RequestBody - Parse JSON request body
