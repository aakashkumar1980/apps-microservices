📊 DATABASE - SQL vs NoSQL Comparison
1. Tables (SQL) vs Collections (NoSQL)
🔑 Unique Keys for High-Performance Filtering:
• Primary Key (PK) - Unique identifier, auto-indexed, fastest lookup O(1)
• Foreign Key (FK) - Reference to parent table, should be indexed for JOINs
• Unique Index - Ensures uniqueness + fast lookup (e.g., email, username)
• Composite Key - Multiple columns as unique identifier (e.g., offer_id + customer_id)

2. Simple Queries - JOINs
2.1 INNER JOIN - Returns only matching rows from both tables
📋 Use Case: Get all offers WITH their campaigns. Excludes offers without campaigns (O004 with NULL campaign_id). Used for: displaying offer details with campaign info.
SQL - INNER JOIN
/** ═══════════════════════════════════════════════════════════
 *  INNER JOIN: Returns ONLY rows where both tables have matching keys
 *  
 *  Query Execution Order:
 *  1. Use idx_offers_campaign_id to scan offers table
 *  2. For each offer, use idx_campaign_status to lookup campaign
 *  3. Filter WHERE c.status = 'ACTIVE'
 *  4. Return matching rows
 *  ═══════════════════════════════════════════════════════════ */

SELECT 
  c.name,           /** from campaign table */
  o.merchant,       /** from offers table */
  o.discount        /** from offers table */

/** Start with campaign table */
FROM campaign c

/** JOIN offers using FK index idx_offers_campaign_id */
INNER JOIN offers o 
  ON c.id = o.campaign_id

/** Filter using idx_campaign_status */
WHERE c.status = 'ACTIVE';

/** ═══════════════════════════════════════════════════════════
 *  📌 INDEXES REQUIRED (in order of usage):
 *  ═══════════════════════════════════════════════════════════ */

/** 1. Index on FK for JOIN lookup (used first)
 *  - Example: 
 *    SELECT * FROM offers 
 *    WHERE campaign_id = 'C001' */
CREATE INDEX idx_offers_campaign_id ON offers(campaign_id);

/** 2. Index on status for WHERE filtering
 *  - Example: 
 *    SELECT * FROM campaign 
 *    WHERE status = 'ACTIVE' */
CREATE INDEX idx_campaign_status ON campaign(status);

3. Advanced Queries - Aggregations & Window Functions
3.1 GROUP BY Aggregation - COUNT, SUM, AVG, MAX, MIN
📋 Use Case: Generate campaign performance dashboard - total redemptions, revenue, and averages per campaign. Used for: reporting, analytics, KPI tracking.
SQL - GROUP BY
/** ═══════════════════════════════════════════════════════════
 *  GROUP BY: Collapse multiple rows into one per group
 *  
 *  IMPORTANT RULES:
 *  - SELECT can ONLY have: GROUP BY columns OR aggregate functions
 *  - If you need other columns (like name), use outer query
 *  - GROUP BY only needs the unique key (id), not dependent columns
 *  ═══════════════════════════════════════════════════════════ */

/** Outer query to get additional columns like 'name' */
SELECT 
  agg.id,           /** from aggregated result */
  c.name,           /** from campaign (joined) */
  agg.total_count,
  agg.total_amount,
  agg.avg_amount,
  agg.max_amount
FROM (
  /** Inner query: GROUP BY with aggregates
   *  SELECT only allows: GROUP BY columns + aggregate functions */
  SELECT 
    c.id,           /** GROUP BY key (PK is unique) */
    
    /** Aggregate functions - applied to each group */
    COUNT(r.id) AS total_count,       /** count rows */
    SUM(r.amount) AS total_amount,    /** sum amounts */
    AVG(r.amount) AS avg_amount,      /** average */
    MAX(r.amount) AS max_amount       /** maximum */
  
  FROM campaign c
  INNER JOIN offers o ON c.id = o.campaign_id
  INNER JOIN redemption r ON o.id = r.offer_id
  
  /** GROUP BY only needs unique key (id)
   *  No need to include 'name' - it depends on id */
  GROUP BY c.id
  
  /** HAVING filters AFTER grouping (unlike WHERE) */
  HAVING COUNT(r.id) > 1
  
) agg
/** Join back to get 'name' column */
INNER JOIN campaign c ON agg.id = c.id;

/** ═══════════════════════════════════════════════════════════
 *  📌 INDEXES REQUIRED (in order of usage):
 *  ═══════════════════════════════════════════════════════════ */

/** 1. FK index for offers → campaign JOIN
 *  - Example: 
 *    SELECT * FROM offers 
 *    WHERE campaign_id = 'C001' */
CREATE INDEX idx_offers_campaign_id ON offers(campaign_id);

/** 2. FK index for redemption → offers JOIN
 *  - Example: 
 *    SELECT * FROM redemption 
 *    WHERE offer_id = 'O001' */
CREATE INDEX idx_redemption_offer_id ON redemption(offer_id);

3.2 PARTITION BY + ROW_NUMBER - Top N per Group
📋 Use Case: Find the TOP redemption (highest amount) for EACH campaign. Useful for: identifying best-performing offers, leaderboard per campaign.
SQL - ROW_NUMBER() with PARTITION BY
/** ═══════════════════════════════════════════════════════════
 *  WINDOW FUNCTION: ROW_NUMBER() with PARTITION BY
 *  - PARTITION BY: Divides data into groups (like GROUP BY)
 *  - ORDER BY: Sorts within each partition
 *  - ROW_NUMBER(): Assigns 1, 2, 3... within each partition
 *  ═══════════════════════════════════════════════════════════ */

SELECT * FROM (

  /** Step 1: JOIN all tables to get redemptions with campaign info */
  SELECT 
    c.id AS campaign_id,
    c.name AS campaign,
    o.merchant,
    r.amount,
    
    /** Step 2: Apply window function */
    ROW_NUMBER() OVER (
    
      /** Step 2.1: PARTITION BY - create separate groups by campaign
       *  Each campaign gets its own numbering starting from 1 */
      PARTITION BY c.id
      
      /** Step 2.2: ORDER BY - sort rows WITHIN each partition
       *  DESC = highest amount gets row_number = 1 */
      ORDER BY r.amount DESC
      
    ) AS rn  /** Step 2.3: assigns 1,2,3... within each group */
    
  FROM campaign c
  JOIN offers o ON c.id = o.campaign_id
  JOIN redemption r ON o.id = r.offer_id
  
) ranked

/** Step 3: Filter to keep only TOP row (rn=1) per campaign */
WHERE rn = 1;

/** ═══════════════════════════════════════════════════════════
 *  📌 INDEXES REQUIRED:
 *  ═══════════════════════════════════════════════════════════ */

/** 1. FK index for offers → campaign JOIN
 *  - Example: 
 *    SELECT * FROM offers 
 *    WHERE campaign_id = 'C001' */
CREATE INDEX idx_offers_campaign_id ON offers(campaign_id);

/** 2. FK index for redemption → offers JOIN
 *  - Example: 
 *    SELECT * FROM redemption 
 *    WHERE offer_id = 'O001' */
CREATE INDEX idx_redemption_offer_id ON redemption(offer_id);

NoSQL - N1QL Window Functions (Couchbase)
/** ═══════════════════════════════════════════════════════════
 *  Couchbase N1QL (7.0+ supports window functions)
 *  - Same syntax as SQL window functions
 *  - PARTITION BY + ORDER BY within OVER clause
 *  ═══════════════════════════════════════════════════════════ */

SELECT * FROM (
  /** Inner query: JOIN + Window Function */
  SELECT 
    c.id AS campaign_id,  /** from campaign */
    c.name,               /** from campaign */
    o.merchant,           /** from offers */
    r.amount,             /** from redemption */
    
    /** Window function: assigns row numbers within partition */
    ROW_NUMBER() OVER (
      PARTITION BY c.id   /** group by campaign */
      ORDER BY r.amount DESC  /** highest first */
    ) AS rn              /** rn=1 is top per group */
    
  FROM campaign c
  JOIN offers o ON o.campaign_id = c.id
  JOIN redemption r ON r.offer_id = o.id
  WHERE c.type = 'campaign'   /** type filter */
    AND o.type = 'offer'       /** type filter */
    AND r.type = 'redemption'  /** type filter */
) ranked
/** Filter to keep only TOP row per campaign */
WHERE rn = 1;

/** ═══════════════════════════════════════════════════════════
 *  📌 INDEXES REQUIRED (in order of usage):
 *  ═══════════════════════════════════════════════════════════ */

/** 1. Index on FK campaign_id for offers → campaign JOIN
 *  - Example: 
 *    SELECT * FROM offers

4. Java 8 Streams - Equivalent Operations
⚠️ Best Practice: Use Java Streams for small datasets (<10K) or complex transformations. For large data, prefer database queries with proper indexes.
☕ INNER JOIN (flatMap + filter)
/** ═══════════════════════════════════════════════════════════
 *  Simulate INNER JOIN using flatMap + filter
 *  
 *  flatMap purpose: Collects nested objects into one flat list
 *  - Each campaign has multiple offers (1:N relationship)
 *  - flatMap flattens: [[O001,O002], [O003]] → [O001,O002,O003]
 *  - Without flatMap: would get List<List<Offer>> instead of List<Offer>
 *  ═══════════════════════════════════════════════════════════ */

List<CampaignOfferDTO> result = 
  campaigns.stream()
  
    /** Step 1: Filter ACTIVE campaigns
     *  - filter() keeps elements matching predicate
     *  - Input:  [C001(ACTIVE), C002(DRAFT), C003(ACTIVE)]
     *  - Output: [C001, C003] (only ACTIVE) */
    .filter(c -> "ACTIVE"
        .equals(c.getStatus()))
    
    /** Step 2: flatMap - collect nested offers into one list
     *  - For each campaign, find matching offers
     *  - C001 has offers: [O001, O002]
     *  - C003 has offers: [O003]
     *  - flatMap flattens into: [O001, O002, O003]
     *  - This is like SQL: FROM campaign JOIN offers */
    .flatMap(c -> 
      offers.stream()
        /** Filter offers by campaign_id (JOIN condition) */
        .filter(o -> c.getId()
            .equals(o.getCampaignId()))
        /** Step 3: Transform to DTO (like SELECT) */
        .map(o -> new DTO(
            c.getName(), 
            o.getMerchant()))
    )
    /** Step 4: Collect stream into List */
    .collect(Collectors.toList());

☕ GROUP BY (groupingBy)
/** ═══════════════════════════════════════════════════════════
 *  Simulate GROUP BY with aggregations
 *  
 *  groupingBy purpose: Groups elements by a key
 *  - Similar to SQL GROUP BY clause
 *  - Returns Map<Key, List<Values>> or Map<Key, AggregateResult>
 *  ═══════════════════════════════════════════════════════════ */

Map<String, Stats> stats = 
  redemptions.stream()
  
  /** Step 1: Group by campaign_id (like SQL GROUP BY)
   *  - Creates buckets: key → list of values
   *  - C001 → [R001($50), R002($75), R003($100)]
   *  - C003 → [R004($200)] */
  .collect(Collectors.groupingBy(
    Redemption::getCampaignId,
    
    /** Step 2: For each group, apply downstream collector
     *  - collectingAndThen: collect first, then transform
     *  - toList() collects group into List
     *  - Lambda calculates aggregates on the List */
    Collectors.collectingAndThen(
      Collectors.toList(),
      list -> {
        /** COUNT - same as SQL COUNT(*) */
        int count = list.size();
        
        /** SUM - same as SQL SUM(amount) */
        double sum = list.stream()
          .mapToDouble(R::getAmount)
          .sum();
        
        /** AVG - same as SQL AVG(amount) */
        double avg = sum / count;
        
        return new Stats(count, sum, avg);
      }
    )
  ));

☕ TOP N per Group (maxBy)
/** ═══════════════════════════════════════════════════════════
 *  Simulate PARTITION BY + TOP 1 (Window Function)
 *  
 *  maxBy purpose: Finds maximum element by comparator
 *  - Similar to SQL: ROW_NUMBER() OVER (PARTITION BY ... ORDER BY) WHERE rn=1
 *  - Returns Optional<T> (may be empty if group is empty)
 *  ═══════════════════════════════════════════════════════════ */

Map<String, Redemption> topPerCampaign = 
  redemptions.stream()
  
  /** Step 1: Group by campaign_id (like PARTITION BY)
   *  - Creates partitions for each campaign
   *  - C001 → [R001($50), R002($75), R003($100)]
   *  - C003 → [R004($200)] */
  .collect(Collectors.groupingBy(
    Redemption::getCampaignId,
    
    /** Step 2: Find MAX in each group (like ORDER BY + TOP 1)
     *  - maxBy finds the element with highest value
     *  - Comparator.comparing extracts the comparison key */
    Collectors.collectingAndThen(
      /** maxBy returns Optional<Redemption>
       *  - Compares by amount field
       *  - Returns element with highest amount */
      Collectors.maxBy(
        Comparator.comparing(
          Redemption::getAmount)),
      
      /** Unwrap Optional to get actual value
       *  - orElse(null) handles empty groups */
      opt -> opt.orElse(null)
    )
  ));

/** Result: Top redemption per campaign
 *  - C001 → R003 ($100) - highest in group
 *  - C003 → R004 ($200) - only one in group */
